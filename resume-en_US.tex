\section{Research Interests}
My research focuses on \textbf{Retrieval-Augmented Generation (RAG)} and \textbf{LLM Agent Training}, with two core directions:
\textbf{(1) RAG \& Deep Search}---building reliable, adaptive, and abstention-aware retrieval-augmented systems that know when to retrieve, what to retrieve, how to retrieve, and when to stop or refuse, toward autonomous deep research agents;
\textbf{(2) LLM Agent Training}---training LLM-based agents through reinforcement learning and post-training to perform multi-turn exploration, tool use, structured reasoning, fine-grained credit assignment, and learning from environment feedback.
I am currently an \textbf{Ant Star Intern} with the Machine Intelligence Division at Ant Group.
During my Ph.D. study, I have published \textbf{6 first-/co-first-author papers in top-tier conferences/journals (4 CCF-A and 2 CCF-B)}, with \textbf{3 additional first-/co-first-author papers under review} (EMNLP 2026 and AAAI 2027 $\times$ 2).

\section{Publications}

\textbf{\color{primary} $\blacktriangleright$ \quad First-Author Publications}
\begin{enumerate}[leftmargin=0.3in, label={[\arabic*]}]

  \pubentry{https://arxiv.org/abs/2512.10999}{KBQA-R1: Reinforcing Large Language Models for Knowledge Base Question Answering}
  {\textbf{\underline{Xin Sun}}, Zhongqi Chen, Xing Zheng, Qiang Liu, Shu Wu, Bowen Song, Zilei Wang, Weiqiang Wang, Liang Wang}
  {ICML 2026 (CCF-A)}{}
  {\small\color{secondary} Formulates knowledge base question answering as a multi-turn MDP where models adapt their reasoning strategies through closed-loop interaction with the KB; introduces RRS for RL cold start and RRGC for sparse rewards, improving accuracy while achieving 26$\times$ higher inference efficiency than MCTS-based methods.}

  \pubentry{https://arxiv.org/abs/2605.29584}{GAPD: Gold-Action Policy Distillation for Agentic Reinforcement Learning in Knowledge Base Question Answering}
  {\textbf{\underline{Xin Sun}}, Zhongqi Chen, Xing Zheng, Qiang Liu, Shu Wu, Bowen Song, Zilei Wang, Weiqiang Wang, Liang Wang}
  {Under Review at EMNLP 2026}{}
  {\small\color{secondary} Addresses sparse supervision in outcome-only RL for agentic KBQA by proposing a gold-action-conditioned self-teacher and Entity-Anchor Matching, converting gold actions into safely aligned token-level policy guidance while preserving the original GRPO objective.}

  \pubentry{https://arxiv.org/abs/2608.01359}{EviSD: Evidence-Conditioned Self-Distillation for Search-Augmented Agents}
  {Jianan Xie\textsuperscript{*}, \textbf{\underline{Xin Sun}}\textsuperscript{*}, Zhongqi Chen, Xing Zheng, Shu Wu, Bowen Song, Liang Wang}
  {Under Review at AAAI 2027 \quad \href{https://arxiv.org/abs/2608.01359}{arXiv:2608.01359} \quad \textsuperscript{*}Equal contribution}{}
  {\small\color{secondary} Proposes evidence-conditioned self-distillation that uses supporting evidence and golden answers as privileged information for action-level credit assignment, achieving the best macro-average EM across seven QA benchmarks while modulating only 6.7\%--15.1\% of response tokens.}

  \pubentry{https://arxiv.org/abs/2608.01358}{HopRefusalBench: Diagnosing Refusal Failures in Search-Augmented Agents for Multi-Hop Reasoning}
  {Jianan Xie\textsuperscript{*}, \textbf{\underline{Xin Sun}}\textsuperscript{*}, Zhongqi Chen, Xing Zheng, Qiang Liu, Bowen Song}
  {Under Review at AAAI 2027 \quad \href{https://arxiv.org/abs/2608.01358}{arXiv:2608.01358} \quad \textsuperscript{*}Equal contribution}{}
  {\small\color{secondary} Introduces an 889-question benchmark crossing three causes of unanswerability with root, middle, and terminal topologies; evaluation of ten frontier models finds a best target-aware correct halting rate of only 42.9\%, exposing persistent hallucination and search-budget exhaustion.}

  \pubentry{https://aclanthology.org/2025.acl-long.561/}{Divide-Then-Align: Honest Alignment based on the Knowledge Boundary of RAG}
  {\textbf{\underline{Xin Sun}}, Jianan Xie, Zhongqi Chen, Qiang Liu, Shu Wu, Yuehe Chen, Bowen Song, Zilei Wang, Weiqiang Wang, Liang Wang}
  {ACL 2025, Main Conference (CCF-A)}{}
  {\small\color{secondary} Defines a four-quadrant knowledge boundary for RAG based on parametric and retrieved knowledge, constructs quadrant-specific preference data, and performs honest alignment with DPO, SFT, and auxiliary classification objectives to teach models to refuse when knowledge is insufficient.}

  \pubentry{https://arxiv.org/abs/2601.11443}{Predict the Retrieval! Test Time Adaptation for Retrieval Augmented Generation}
  {\textbf{\underline{Xin Sun}}, Zhongqi Chen, Qiang Liu, Shu Wu, Bowen Song, Weiqiang Wang, Zilei Wang, Liang Wang}
  {ICASSP 2026 (CCF-B)}{}
  {\small\color{secondary} Constructs self-supervised prefix-suffix prediction tasks from retrieved documents and dynamically updates model parameters at inference time, enabling RAG systems to adapt to specialized domains without labeled data.}

  \pubentry{https://arxiv.org/abs/2408.04400}{DIVE: Subgraph Disagreement for Graph Out-of-Distribution Generalization}
  {\textbf{\underline{Xin Sun}}, Liang Wang, Qiang Liu, Shu Wu, Zilei Wang, Liang Wang}
  {KDD 2024 (CCF-A)}{}
  {\small\color{secondary} Learns environment-invariant graph representations via subgraph disagreement, improving graph neural network generalization under distribution shifts.}

  \pubentry{https://aclanthology.org/2023.findings-emnlp.876/}{Noise-Robust Semi-Supervised Learning for Distant Supervision Relation Extraction}
  {\textbf{\underline{Xin Sun}}, Qiang Liu, Shu Wu, Zilei Wang, Liang Wang}
